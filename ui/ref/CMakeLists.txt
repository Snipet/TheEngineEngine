# ui/ref/CMakeLists.txt: RefPanel, the reference engine's own panel (sim/ref/panel.gui and its shaders) in our
# window, and tee::ui::makeMainPanel (card J, item 4).
#
# The panel's files are embedded when the application is built (RefFiles.cpp, generated here from sim/ref/): the .gui
# file as text, and each shader's source as text too (the panel reads it for whether the shader moves with time, and
# for a note's sake); the shaders' binaries come from cmake/TeeShaders.cmake. Until the keel puts sim/ref/ in the tree
# the embedded set is empty and RefPanel says so on its face.
set(_ref_dir "${TEE_UI_REPO}/sim/ref")
set(_ref_names panel.gui header.sc cluster.sc layout.sc)
set(_ref_files "")
set(_ref_shaders "")
set(_ref_have ON)
foreach(_n IN LISTS _ref_names)
  if(NOT EXISTS "${_ref_dir}/${_n}")
    set(_ref_have OFF)
  endif()
endforeach()
file(GLOB _ref_headers CONFIGURE_DEPENDS "${_ref_dir}/*.sh")

set(_ref_cpp "// Generated by ui/ref/CMakeLists.txt from sim/ref/: do not edit.\n#include \"ref/RefPanel.h\"\n\n")
string(APPEND _ref_cpp "namespace tee::ui\n{\n    Sources refPanelFiles()\n    {\n        Sources files;\n")
if(_ref_have)
  foreach(_f IN LISTS _ref_names _ref_headers)
    if(IS_ABSOLUTE "${_f}")
      set(_path "${_f}")
      get_filename_component(_n "${_f}" NAME)
    else()
      set(_path "${_ref_dir}/${_f}")
      set(_n "${_f}")
    endif()
    file(READ "${_path}" _text)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_path}")
    # A raw string literal a piece (MSVC's limit on one literal is 16 kB): cut at line ends, about 8 kB each.
    string(APPEND _ref_cpp "        files.push_back({ \"${_n}\", std::string()\n")
    string(LENGTH "${_text}" _len)
    set(_at 0)
    while(_at LESS _len)
      math(EXPR _to "${_at} + 8000")
      if(_to LESS _len)
        string(SUBSTRING "${_text}" ${_to} -1 _tail)
        string(FIND "${_tail}" "\n" _nl)
        if(_nl LESS 0)
          set(_to ${_len})
        else()
          math(EXPR _to "${_to} + ${_nl} + 1")
        endif()
      else()
        set(_to ${_len})
      endif()
      math(EXPR _n_chars "${_to} - ${_at}")
      string(SUBSTRING "${_text}" ${_at} ${_n_chars} _piece)
      string(APPEND _ref_cpp "            + R\"TEEREF(${_piece})TEEREF\"\n")
      set(_at ${_to})
    endwhile()
    string(APPEND _ref_cpp "            });\n")
    if(_n MATCHES "\\.sc$")
      list(APPEND _ref_shaders "${_path}")
    endif()
  endforeach()
else()
  message(STATUS "ui/ref: sim/ref/ is not in the tree yet: RefPanel has no files in this build")
endif()
string(APPEND _ref_cpp "        return files;\n    }\n}\n")
tee_write_if_changed("${CMAKE_CURRENT_BINARY_DIR}/RefFiles.cpp" "${_ref_cpp}")

add_library(tee_ui_ref STATIC RefPanel.cpp RefPanel.h Strip.cpp Strip.h MainPanel.cpp
                              "${CMAKE_CURRENT_BINARY_DIR}/RefFiles.cpp")
target_link_libraries(tee_ui_ref PUBLIC tee_ui_kit)
tee_ui_warnings(tee_ui_ref)

if(_ref_have)
  tee_add_panel_shaders(tee_ui_ref NAME ref GUI "${_ref_dir}/panel.gui" SHADERS ${_ref_shaders} HEADERS ${_ref_headers})
else()
  tee_add_panel_shaders(tee_ui_ref NAME ref GUI "${_ref_dir}/panel.gui")
endif()
